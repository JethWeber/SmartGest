    _db.LancamentoDetalhes.RemoveRange(anteriores);

    else
    {

        Adicionar(lanc, contaOrigem.Id,   0, lanc.Valor);

        Adicionar(lanc, contaEstado.Id,  0,   is_);
    }

// Novo — resolve a conta PGC a partir do banco registado
private async Task<ContaContabil> ResolverContaOrigemAsync(int? contaBancariaId)
{
    if (contaBancariaId.HasValue)
    {
        var banco = await _db.ContasBancarias
            .Include(c => c.ContaContabil)
            .FirstOrDefaultAsync(c => c.Id == contaBancariaId.Value);

        if (banco?.ContaContabil is not null)
            return banco.ContaContabil;
    }

    // Fallback seguro: Caixa e Equivalentes (conta 11)
    return await _db.ContasContabeis.FirstAsync(c => c.Codigo == "11");
}

// Infrastructure/SmartGestDbContext.cs
public class SmartGestDbContext(DbContextOptions<SmartGestDbContext> options)
    : DbContext(options)
{
    public DbSet<Lancamento> Lancamentos => Set<Lancamento>();
    public DbSet<MovimentoContabil> MovimentosContabeis => Set<MovimentoContabil>();

    protected override void OnModelCreating(ModelBuilder mb)
    {
        mb.Entity<Lancamento>(e =>
        {
            e.ToTable("lancamentos");
            e.HasKey(x => x.Id);
            e.Property(x => x.Valor).HasPrecision(18, 2);
            e.Property(x => x.ImpostoSeloCalculado).HasPrecision(18, 2);
            e.HasMany(x => x.Movimentos)
             .WithOne(x => x.Lancamento)
             .HasForeignKey(x => x.LancamentoId);
        });

        mb.Entity<MovimentoContabil>(e =>
        {
            e.ToTable("movimentos_contabeis");
            e.HasKey(x => x.Id);
            e.Property(x => x.Debito).HasPrecision(18, 2);
            e.Property(x => x.Credito).HasPrecision(18, 2);
        });
    }

// Application/MotorContabilService.cs
public class MotorContabilService
{
    public void Processar(Lancamento lancamento)
    {
        var contaOrigem = MapearOrigem(lancamento.OrigemDinheiro);
        var contaCategoria = MapearCategoria(lancamento.Categoria);
        bool geraIS = GeraImpostoDeSelo(lancamento.Categoria);

        // Partidas principais
        lancamento.Movimentos.Add(Movimento(lancamento.Id, contaOrigem,   lancamento.Valor, 0));
        lancamento.Movimentos.Add(Movimento(lancamento.Id, contaCategoria, 0, lancamento.Valor));

        // Imposto de Selo
        if (geraIS)
        {
            decimal is_ = Math.Round(lancamento.Valor * 0.01m, 2);
            lancamento.ImpostoSeloCalculado = is_;
            lancamento.Movimentos.Add(Movimento(lancamento.Id, "69.1", is_, 0));
            lancamento.Movimentos.Add(Movimento(lancamento.Id, "34.5", 0,   is_));
        }
    }

    private static MovimentoContabil Movimento(Guid id, string conta, decimal db, decimal cr) =>
        new() { LancamentoId = id, ContaPGC = conta, Debito = db, Credito = cr };

    private static string MapearCategoria(string c) => c switch
    {
        "Venda de Mercadorias"  => "61.1",
        "Prestação de Serviços" => "62.1",
        "Renda do Espaço"       => "75.2.1",
        "Pagamento de Salários" => "72.2",
        _                       => "78.9"
    };

    private static string MapearOrigem(string o) => o switch
    {
        "Banco BFA" => "12.1.2",
        "Banco BAI" => "12.1.1",
        _           => "11.1"
    };

    private static bool GeraImpostoDeSelo(string c) =>
        c is "Venda de Mercadorias" or "Prestação de Serviços";
}

using Microsoft.EntityFrameworkCore;
using SmartGest.API.Data;
using SmartGest.API.Models;

namespace SmartGest.API.Services;

public class MotorContabilService
{
    private readonly AppDbContext _db;
    public MotorContabilService(AppDbContext db) => _db = db;

    // ── Ponto de entrada ──────────────────────────────────────────────────────
    public async Task ProcessarAsync(Lancamento lanc)
    {
        // Apaga detalhes anteriores (útil no Atualizar)
        var anteriores = _db.LancamentoDetalhes.Where(d => d.LancamentoId == lanc.Id);
        _db.LancamentoDetalhes.RemoveRange(anteriores);

        var contaMovimento  = await ResolverContaAsync(CodigoOrigem(lanc.OrigemDinheiro));
        var contaCategoria  = await ResolverContaAsync(CodigoCategoria(lanc.Categoria));

        // Partidas principais
        if (lanc.Tipo == "Entrada")
        {
            // Débito: disponibilidades (dinheiro entrou)
            // Crédito: receita (motivo)
            Adicionar(lanc, contaMovimento.Id,  lanc.Valor, 0);
            Adicionar(lanc, contaCategoria.Id,  0, lanc.Valor);
        }
        else
        {
            // Débito: despesa (motivo)
            // Crédito: disponibilidades (dinheiro saiu)
            Adicionar(lanc, contaCategoria.Id,  lanc.Valor, 0);
            Adicionar(lanc, contaMovimento.Id,  0, lanc.Valor);
        }

        // Imposto de Selo — apenas receitas operacionais
        if (GeraImpostoDeSelo(lanc.Categoria))
        {
            var is_ = Math.Round(lanc.Valor * 0.01m, 2);
            lanc.ImpostoSelo = is_;

            var contaEncargo = await ResolverContaAsync("65"); // Impostos e Taxas
            var contaEstado  = await ResolverContaAsync("24"); // Imposto a Pagar
            Adicionar(lanc, contaEncargo.Id, is_, 0);
            Adicionar(lanc, contaEstado.Id,  0, is_);
        }
        else
        {
            lanc.ImpostoSelo = 0;
        }
    }

    // ── Mapeamentos ───────────────────────────────────────────────────────────
    private static string CodigoCategoria(string categoria) => categoria switch
    {
        "Venda de Mercadorias"  => "71",
        "Prestação de Serviços" => "72",
        "Renda do Espaço"       => "73",
        "Pagamento de Salários" => "63",
        "Fornecimentos/Serviços Externos" => "62",
        "Outros Custos"         => "66",
        _                       => "73"  // Outros Rendimentos Operacionais
    };

    private static string CodigoOrigem(string origem) => origem switch
    {
        "Banco BFA" or "Banco BAI" or "Banco BIC"
        or "Banco BPC" or "Banco ATL" => "11", // Caixa e Equivalentes
        "Caixa Geral"                 => "11",
        _                             => "11"
    };

    private static bool GeraImpostoDeSelo(string categoria) =>
        categoria is "Venda de Mercadorias" or "Prestação de Serviços";

    // ── Helpers ───────────────────────────────────────────────────────────────
    private async Task<ContaContabil> ResolverContaAsync(string codigoPrefixo)
    {
        var conta = await _db.ContasContabeis
            .Where(c => c.Activa && c.Codigo == codigoPrefixo)
            .FirstOrDefaultAsync();

        // Fallback: conta genérica de Outros Rendimentos
        return conta ?? await _db.ContasContabeis.FirstAsync(c => c.Codigo == "73");
    }

    private void Adicionar(Lancamento lanc, int contaId, decimal debito, decimal credito)
    {
        _db.LancamentoDetalhes.Add(new LancamentoDetalhe
        {
            LancamentoId    = lanc.Id,
            ContaContabilId = contaId,
            Debito          = debito,
            Credito         = credito
        });
    }
}

using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using SmartGest.API.Data;
using SmartGest.API.DTOs.Requests;
using SmartGest.API.DTOs.Responses;
using SmartGest.API.Models;
using SmartGest.API.Services;

namespace SmartGest.API.Controllers;

[ApiController]
[Route("api/lancamentos")]
[Authorize]
public class LancamentosController : ControllerBase
{
    private readonly AppDbContext        _db;
    private readonly MotorContabilService _motor;

    public LancamentosController(AppDbContext db, MotorContabilService motor)
    {
        _db    = db;
        _motor = motor;
    }

    // ── GET /api/lancamentos ──────────────────────────────────────────────────
    [HttpGet]
    public async Task<IActionResult> Listar(
        [FromQuery] string?   tipo       = null,
        [FromQuery] DateTime? dataInicio = null,
        [FromQuery] DateTime? dataFim    = null,
        [FromQuery] string?   texto      = null,
        [FromQuery] int?      contaId    = null,
        [FromQuery] int       pagina     = 1,
        [FromQuery] int       tamPagina  = 50)
    {
        var query = _db.Lancamentos.Include(l => l.ContaBancaria).AsQueryable();

        if (!string.IsNullOrWhiteSpace(tipo))  query = query.Where(l => l.Tipo == tipo);
        if (dataInicio.HasValue)               query = query.Where(l => l.Data >= dataInicio.Value);
        if (dataFim.HasValue)                  query = query.Where(l => l.Data <= dataFim.Value);
        if (!string.IsNullOrWhiteSpace(texto)) query = query.Where(l =>
            l.Descricao.Contains(texto) || l.Categoria.Contains(texto) || l.Beneficiario.Contains(texto));
        if (contaId.HasValue)                  query = query.Where(l => l.ContaBancariaId == contaId);

        var total = await query.CountAsync();
        var items = await query
            .OrderByDescending(l => l.Data)
            .Skip((pagina - 1) * tamPagina)
            .Take(tamPagina)
            .Select(l => Map(l))
            .ToListAsync();

        return Ok(new { total, pagina, tamPagina, items });
    }

    // ── GET /api/lancamentos/{id} ─────────────────────────────────────────────
    [HttpGet("{id}")]
    public async Task<IActionResult> Obter(int id)
    {
        var l = await _db.Lancamentos
            .Include(x => x.ContaBancaria)
            .FirstOrDefaultAsync(x => x.Id == id);
        return l is null ? NotFound() : Ok(Map(l));
    }

    // ── POST /api/lancamentos ─────────────────────────────────────────────────
    [HttpPost]
    public async Task<IActionResult> Criar([FromBody] LancamentoRequest req)
    {
        var lanc = new Lancamento
        {
            Data              = req.Data,
            Descricao         = req.Descricao,
            Categoria         = req.Categoria,
            Tipo              = req.Tipo,
            Valor             = req.Valor,
            Beneficiario      = req.Beneficiario,
            MetodoPagamento   = req.MetodoPagamento,
            CaminhoDocumento  = req.CaminhoDocumento,
            Observacoes       = req.Observacoes,
            CentroCusto       = req.CentroCusto,
            ReferenciaInterna = req.ReferenciaInterna,
            OrigemDinheiro    = req.OrigemDinheiro,
            ContaBancariaId   = req.ContaBancariaId,
        };

        _db.Lancamentos.Add(lanc);
        await _db.SaveChangesAsync(); // Precisa do Id antes do motor

        // Motor contábil — gera LancamentoDetalhes e calcula ImpostoSelo
        await _motor.ProcessarAsync(lanc);

        // Actualiza saldo da conta bancária
        await AtualizarSaldoAsync(req.ContaBancariaId, req.Tipo, req.Valor);

        await _db.SaveChangesAsync();

        return CreatedAtAction(nameof(Obter), new { id = lanc.Id }, Map(lanc));
    }

    // ── PUT /api/lancamentos/{id} ─────────────────────────────────────────────
    [HttpPut("{id}")]
    public async Task<IActionResult> Atualizar(int id, [FromBody] LancamentoRequest req)
    {
        var lanc = await _db.Lancamentos.FindAsync(id);
        if (lanc is null) return NotFound();

        // Reverter o efeito no saldo bancário anterior
        await AtualizarSaldoAsync(lanc.ContaBancariaId,
            lanc.Tipo == "Entrada" ? "Saída" : "Entrada", lanc.Valor);

        lanc.Data              = req.Data;
        lanc.Descricao         = req.Descricao;
        lanc.Categoria         = req.Categoria;
        lanc.Tipo              = req.Tipo;
        lanc.Valor             = req.Valor;
        lanc.Beneficiario      = req.Beneficiario;
        lanc.MetodoPagamento   = req.MetodoPagamento;
        lanc.CaminhoDocumento  = req.CaminhoDocumento;
        lanc.Observacoes       = req.Observacoes;
        lanc.CentroCusto       = req.CentroCusto;

        lanc.OrigemDinheiro    = req.OrigemDinheiro;
        lanc.ContaBancariaId   = req.ContaBancariaId;

        // Re-processa o motor (apaga os detalhes antigos e regera)
        await _motor.ProcessarAsync(lanc);

        // Aplica novo efeito no saldo
        await AtualizarSaldoAsync(req.ContaBancariaId, req.Tipo, req.Valor);

        await _db.SaveChangesAsync();
        return Ok(Map(lanc));
    }

    // ── DELETE /api/lancamentos/{id} ──────────────────────────────────────────
    [HttpDelete("{id}")]
    [Authorize(Roles = "Administrador,Contabilista")]
    public async Task<IActionResult> Eliminar(int id)
    {
        var lanc = await _db.Lancamentos.FindAsync(id);
        if (lanc is null) return NotFound();

        // Reverter saldo antes de apagar
        await AtualizarSaldoAsync(lanc.ContaBancariaId,
            lanc.Tipo == "Entrada" ? "Saída" : "Entrada", lanc.Valor);

        _db.Lancamentos.Remove(lanc); // Cascade apaga os LancamentoDetalhes
        await _db.SaveChangesAsync();
        return NoContent();
    }

    private async Task AtualizarSaldoAsync(int? contaId, string tipo, decimal valor)
    {

        if (conta is null) return;
        conta.SaldoOntem = conta.SaldoAtual;
        conta.SaldoAtual += tipo == "Entrada" ? valor : -valor;

public async Task ProcessarAsync(Lancamento lanc)
{
    var anteriores = _db.LancamentoDetalhes.Where(d => d.LancamentoId == lanc.Id);
    _db.LancamentoDetalhes.RemoveRange(anteriores);

    // Conta da categoria — ainda via mapeamento (poucos e estáveis)
    var contaCategoria = await ResolverContaAsync(CodigoCategoria(lanc.Categoria));

    // Conta da origem — vem do banco registado pelo utilizador
    var contaOrigem = await ResolverContaOrigemAsync(lanc.ContaBancariaId);

    if (lanc.Tipo == "Entrada")
    {
        Adicionar(lanc, contaOrigem.Id,   lanc.Valor, 0);
        Adicionar(lanc, contaCategoria.Id, 0, lanc.Valor);
    }
    else
    {
        Adicionar(lanc, contaCategoria.Id, lanc.Valor, 0);
        Adicionar(lanc, contaOrigem.Id,   0, lanc.Valor);
    }

    if (GeraImpostoDeSelo(lanc.Categoria))
    {
        var is_         = Math.Round(lanc.Valor * 0.01m, 2);
        lanc.ImpostoSelo = is_;
        var contaEncargo = await ResolverContaAsync("65");
        var contaEstado  = await ResolverContaAsync("24");
        Adicionar(lanc, contaEncargo.Id, is_, 0);
        Adicionar(lanc, contaEstado.Id,  0,   is_);
    }
    else { lanc.ImpostoSelo = 0; }
}

// Novo — resolve a conta PGC a partir do banco registado
private async Task<ContaContabil> ResolverContaOrigemAsync(int? contaBancariaId)
{
    if (contaBancariaId.HasValue)
    {
        var banco = await _db.ContasBancarias
            .Include(c => c.ContaContabil)
            .FirstOrDefaultAsync(c => c.Id == contaBancariaId.Value);

        if (banco?.ContaContabil is not null)
            return banco.ContaContabil;
    }

    // Fallback seguro: Caixa e Equivalentes (conta 11)
    return await _db.ContasContabeis.FirstAsync(c => c.Codigo == "11");
}
